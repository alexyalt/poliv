```python
Также добавляет колонки расхода и аварийной блокировки в controllers,
если их ещё нет.
"""

from __future__ import annotations

import json
import sqlite3


DEFAULT_FLOW_SETTINGS = {
    "flow.enabled_default": True,
    "flow.no_flow_delay_sec": 10,
    "flow.overflow_percent": 150,
    "flow.underflow_percent": 70,

DEFAULT_NOTIFICATION_SETTINGS = {

def _ensure_column(conn, table: str, column: str, col_type: str) -> None:
    """Безопасное добавление колонки, если её ещё нет."""
    cols = [r[1] for r in conn.execute(f"PRAGMA table_info({table})").fetchall()]
    if column not in cols:
        conn.execute(f"ALTER TABLE {table} ADD COLUMN {column} {col_type}")


def migrate(conn: sqlite3.Connection) -> None:
    # --- Добавляем колонки в controllers, если их нет ---
```

## Файл 2. Эмулятор: имитация потока и аварий

**`emulator/flow_sim.py`**

> Этот модуль встраивается в существующий эмулятор. Он отслеживает состояние «поливается ли зона», генерирует показания расходомера и умеет имитировать аварии `no_flow`, `over_flow`, `under_flow`.

```python
"""Этап 6: имитация расходомера и аварий потока для эмулятора.

Подключение:
    в основном файле эмулятора (например, fleet_sim.py) для каждого
    контроллера создаётся экземпляр FlowSim и вызывается его tick()
    каждую секунду (или с нужным шагом).

Модуль умеет:
- генерировать показания потока во время полива;
- имитировать аварии по команде/сценарию;
- публиковать события аварий и останавливать полив.
"""

from __future__ import annotations

import json
import random
import time
import uuid


class FlowSim:
    """Имитация расходомера для одного контроллера."""

    def __init__(self, box_id: str, publish_fn, expected_flow_lpm: float = 5.0):
        """
        box_id      — идентификатор контроллера;
        publish_fn  — функция публикации в MQTT: (topic, payload_dict);
        expected_flow_lpm — ожидаемый расход л/мин для расчёта.
        """
        self.box_id = box_id
        self.publish = publish_fn
        self.expected_lpm = expected_flow_lpm

        self.watering_zone = None          # номер зоны, которая поливается
        self.watering_start_ts = None
        self.total_liters = 0.0
        self.instant_lpm = 0.0

        # Аварийное состояние
        self.emergency = None              # None / 'no_flow' / 'over_flow' / 'under_flow'
        self._no_flow_seconds = 0

        # Последний опубликованный поток (чтобы не слать каждую секунду)
        self._last_flow_ts = 0

    # ------------------------------------------------------------------
    # Управление поливом
    # ------------------------------------------------------------------
    def start_watering(self, zone: int) -> None:
        """Начать полив зоны."""
        self.watering_zone = zone
        self.watering_start_ts = int(time.time())
        self.emergency = None
        self._no_flow_seconds = 0

    def stop_watering(self) -> None:
        """Завершить полив зоны и отправить событие."""
        if self.watering_zone is None:
            return
        zone = self.watering_zone
        start_ts = self.watering_start_ts
        end_ts = int(time.time())
        water_sec = max(1, end_ts - start_ts)
        volume = round(self.instant_lpm * water_sec / 60.0, 2) if self.instant_lpm else 0.0

        self._publish_event(
            status="ok",
            zone=zone,
            start_ts=start_ts,
            end_ts=end_ts,
            water_sec=water_sec,
            volume_liters=volume,
        )

        self.watering_zone = None
        self.watering_start_ts = None
        self.instant_lpm = 0.0

    # ------------------------------------------------------------------
    # Аварии
    # ------------------------------------------------------------------
    def trigger_emergency(self, kind: str) -> None:
        """Принудительно создать аварию: 'no_flow' / 'over_flow' / 'under_flow'."""
        if self.watering_zone is None:
            return
        self.emergency = kind
        zone = self.watering_zone
        start_ts = self.watering_start_ts
        end_ts = int(time.time())
        water_sec = max(1, end_ts - start_ts)

        self._publish_event(
            status=kind,
            zone=zone,
            start_ts=start_ts,
            end_ts=end_ts,
            water_sec=water_sec,
            volume_liters=0.0,
            reason_code=self._reason_code(kind),
        )

        # Останавливаем полив
        self.watering_zone = None
        self.watering_start_ts = None
        self.instant_lpm = 0.0

    def clear_emergency(self) -> None:
        """Сброс аварийной блокировки (по команде clear_error)."""
        self.emergency = None
        self._no_flow_seconds = 0

    def _reason_code(self, kind: str) -> int:
        return {"no_flow": 4001, "over_flow": 4002, "under_flow": 4003}.get(kind, 4000)

    # ------------------------------------------------------------------
    # Тик (вызывать периодически)
    # ------------------------------------------------------------------
    def tick(self, force_no_flow: bool = False) -> None:
        """Один шаг симуляции.

        force_no_flow — имитировать отсутствие потока (для теста аварии).
        """
        now = int(time.time())

        if self.watering_zone is not None:
            if force_no_flow:
                self.instant_lpm = 0.0
                self._no_flow_seconds += 1
            else:
                # Нормальный поток с шумом
                self.instant_lpm = round(self.expected_lpm * random.uniform(0.9, 1.1), 2)
                self._no_flow_seconds = 0
                self.total_liters += self.instant_lpm / 60.0  # за 1 сек
        else:
            self.instant_lpm = 0.0

        # Публикуем поток раз в 10 сек при поливе / раз в 30 сек в простое
        interval = 10 if self.watering_zone is not None else 30
        if now - self._last_flow_ts >= interval:
            self._publish_flow()
            self._last_flow_ts = now

        # Автоавария при устойчивом отсутствии потока (10 сек)
        if self.watering_zone is not None and self._no_flow_seconds >= 10:
            self.trigger_emergency("no_flow")

    # ------------------------------------------------------------------
    # Публикация
    # ------------------------------------------------------------------
    def _publish_flow(self) -> None:
        payload = {
            "protocol_version": 1,
            "ts": int(time.time()),
            "box_id": self.box_id,
            "flow_enabled": True,
            "total_liters": round(self.total_liters, 2),
            "instant_lpm": self.instant_lpm,
            "active_zones": [self.watering_zone] if self.watering_zone is not None else [],
            "status": "ok",
        }
        self.publish(f"poliv/{self.box_id}/flow", payload)

    def _publish_event(
        self,
        status: str,
        zone: int,
        start_ts: int,
        end_ts: int,
        water_sec: int,
        volume_liters: float,
        reason_code: int | None = None,
    ) -> None:
        payload = {
            "protocol_version": 1,
            "event_uid": str(uuid.uuid4()),
            "ts": int(time.time()),
            "box_id": self.box_id,
            "run_id": f"run-{uuid.uuid4().hex[:8]}",
            "active_zones": [zone],
            "primary_zone": zone,
            "source": "schedule",
            "start_ts": start_ts,
            "end_ts": end_ts,
            "water_sec": water_sec,
            "wall_sec": water_sec,
            "volume_liters": volume_liters,
            "expected_flow_lpm": self.expected_lpm,
            "status": status,
            "reason_code": reason_code,
            "schedule_version": 1,
            "buffered": False,
        }
        self.publish(f"poliv/{self.box_id}/event", payload)
```

```javascript
/* Этап 6: виджет уведомлений + звуковой сигнал для критических.
 * Подключается в base.html. Опрашивает /api/notifications/unread-count
 * каждые 30 секунд (Артефакт 0.6 §7.1).
 */
(function () {
  var POLL_MS = 30000;
  var lastCriticalId = null;

  function beep() {
    try {
      var ctx = new (window.AudioContext || window.webkitAudioContext)();
      var osc = ctx.createOscillator();
      var gain = ctx.createGain();
      osc.connect(gain);
      gain.connect(ctx.destination);
      osc.type = 'square';
      osc.frequency.value = 880;
      gain.gain.value = 0.1;
      osc.start();
      osc.stop(ctx.currentTime + 0.4);
    } catch (e) {
      // звук не поддерживается — игнорируем
    }
  }

  function updateBadge() {
    fetch('/api/notifications', { credentials: 'same-origin' })
      .then(function (r) {
        if (!r.ok) throw new Error('HTTP ' + r.status);
        return r.json();
      })
      .then(function (d) {
        var badge = document.getElementById('notif-count');
        if (badge) {
          badge.textContent = d.unread_count > 0 ? d.unread_count : '';
          badge.style.display = d.unread_count > 0 ? 'inline-block' : 'none';
        }

        // Критическое уведомление → звук
        var critical = (d.items || []).filter(function (n) {
          return n.severity === 'critical' && !n.read_at;
        });
        if (critical.length > 0) {
          var topId = critical[0].id;
          if (lastCriticalId !== topId) {
            lastCriticalId = topId;
            beep();
            showBanner(critical[0].message);
          }
        }
      })
      .catch(function () { /* сеть недоступна — молча */ });
  }

  function showBanner(text) {
    var el = document.getElementById('notif-banner');
    if (!el) return;
    el.textContent = '⚠ ' + text;
    el.style.display = 'block';
    setTimeout(function () { el.style.display = 'none'; }, 8000);
  }

  setInterval(updateBadge, POLL_MS);
  updateBadge();
})();
```

```markdown
# Инструкция проверки Этапа 6 (События, расход и аварии)

## Подготовка
1. Применить миграции: `python server/main.py` (миграция 0008 должна примениться автоматически).
2. Запустить сервер: `python server/main.py`.
3. Запустить эмулятор с подключённым модулем `flow_sim.py`.

## Проверка 1. События и дедупликация
1. Запустить полив зоны в эмуляторе.
2. Дождаться завершения полива.
3. Проверить `GET /api/events` — событие должно появиться.
4. Проверить БД: `SELECT COUNT(*) FROM watering_events WHERE event_uid = '<uid>'` — должна быть 1 запись.
5. Повторно отправить то же событие (с тем же `event_uid`) — дубликат не должен появиться.

## Проверка 2. Расход и дневные агрегаты
1. После полива проверить `GET /api/flow/daily`.
2. Проверить БД: `SELECT * FROM flow_daily` — должен появиться объём.

## Проверка 3. Авария `no_flow`
1. В эмуляторе вызвать `flow_sim.trigger_emergency('no_flow')` или имитировать отсутствие потока 10 сек.
2. Проверить, что эмулятор остановил полив.
3. Проверить БД: `SELECT emergency_lock_until_ts, status FROM controllers` — блокировка должна быть установлена.
4. Проверить `GET /api/notifications` — должно появиться критическое уведомление.
5. В браузере должен прозвучать звуковой сигнал (если включён звук).

## Проверка 4. Сброс аварии
1. Войти под `admin` или `operator`.
2. `POST /api/controllers/{id}/clear-error` с телом `{"reason": "тест"}`.
3. Проверить БД: `emergency_lock_until_ts` должен быть `NULL`.
4. Проверить, что эмулятор получил команду `clear_error`.

## Проверка 5. Роли
1. Войти под `viewer`.
2. `POST /api/controllers/{id}/clear-error` — должен вернуть 403.
3. `GET /api/events` — должен вернуть 200.

## Проверка 6. Уведомления в интерфейсе
1. Открыть `/notifications`.
2. Проверить отображение уведомлений.
3. Проверить счётчик непрочитанных в шапке (колокольчик).
4. Проверить кнопку «Прочитать все».

## Критерий готовности Этапа 6
- события попадают в БД;
- объём воды считается;
- авария останавливает эмулятор;
- уведомление появляется;
```
